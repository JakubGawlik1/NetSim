package "Storage" {

    interface IPackageStockpile <<interface>> {
        + push(Package&&) : void
        + empty() : bool {query}
        + size() : size_type {query}
        + {method} /iteratory/
        + ~IPackageStockpile()
    }

    interface IPackageQueue <<interface>> {
        + pop() : Package
        + get_queue_type() : PackageQueueType {query}
    }

    IPackageQueue <|-- IPackageStockpile

    enum PackageQueueType {
        FIFO
        LIFO
    }

    PackageQueueType .. IPackageQueue

    class PackageQueue {
        + PackageQueue(PackageQueueType)
    }

    PackageQueue ..|> IPackageQueue

    class Package {
        + Package()
        + Package(ElementID)
        + Package(Package&&)
        + operator=(Package&&) : Package&
        + get_id() : ElementID {query}
        + ~Package()
    }
}

together {
    class ElementID << (T,orchid) primitive>>
}
@enduml
